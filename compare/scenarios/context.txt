# CONTEXT=SEARCH (RFC 5267 section 4), ESORT (RFC 5267 section 3) and the PARTIAL search return option
# Run with: npm run compare -- --plugin CONTEXT=SEARCH,ESORT,SEARCHRES compare/scenarios/context.txt
# Dovecot does not advertise CONTEXT=SORT, but accepts UPDATE for SORT and sends updates with position 0
SELECT INBOX
SEARCH RETURN (UPDATE) UNSEEN
UID SEARCH RETURN (UPDATE COUNT) UNSEEN
SEARCH RETURN (CONTEXT) ALL
SEARCH RETURN (CONTEXT COUNT) ALL
SEARCH RETURN (UPDATE UPDATE) ALL
SEARCH RETURN (PARTIAL 1:2) ALL
SEARCH RETURN (PARTIAL 2:9) ALL
SEARCH RETURN (PARTIAL 5:9) ALL
SEARCH RETURN (PARTIAL 1:2 ALL) ALL
SEARCH RETURN (PARTIAL 1:2 PARTIAL 1:3) ALL
SEARCH RETURN (PARTIAL 0:2) ALL
SEARCH RETURN (COUNT PARTIAL 1:1 MAX) ALL
SEARCH RETURN (SAVE PARTIAL 2:2 MIN) ALL
FETCH $ UID
# updates of this session
STORE 1 -FLAGS (\Seen)
STORE 2:3 +FLAGS (\Seen)
STORE 2 +FLAGS (\Deleted)
EXPUNGE
# updates from another session
2: SELECT INBOX
2: STORE 1 +FLAGS (\Seen)
2: APPEND INBOX {14}\r\nSubject: new\r\n
!wait 1000
NOOP
CANCELUPDATE "A2"
CANCELUPDATE "A2"
CANCELUPDATE A3
CANCELUPDATE
# extended SORT
SORT RETURN (MIN MAX ALL COUNT) (REVERSE ARRIVAL) UTF-8 ALL
UID SORT RETURN () (SUBJECT) UTF-8 ALL
SORT RETURN (SAVE MIN) (REVERSE ARRIVAL) UTF-8 ALL
FETCH $ UID
SORT RETURN (COUNT) (DATE) UTF-8 DELETED
SORT RETURN (MIN) (DATE) UTF-8 DELETED
