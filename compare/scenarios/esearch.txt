# ESEARCH (RFC 4731), SEARCHRES (RFC 5182) and SEARCH MODSEQ (RFC 7162 section 3.1.5)
# Run with: npm run compare -- --plugin SEARCHRES,CONDSTORE,UIDPLUS compare/scenarios/esearch.txt
# MODSEQ values differ, Dovecot assigns its own mod-sequences when the user is seeded
SELECT INBOX
SEARCH RETURN (MIN MAX COUNT ALL) ALL
UID SEARCH RETURN () ALL
SEARCH RETURN (MIN MAX ALL) DELETED
SEARCH RETURN (COUNT) DELETED
SEARCH RETURN (COUNT) CHARSET UTF-8 ALL
SEARCH RETURN (FOO) ALL
SEARCH RETURN MIN ALL
SEARCH CHARSET UTF-8 RETURN (ALL) ALL
SEARCH RETURN (SAVE) 2:3
FETCH $ UID
UID FETCH $ UID
SEARCH $
UID SEARCH UID $
FETCH 1,$ UID
SEARCH RETURN (SAVE MIN MAX) ALL
FETCH $ UID
SEARCH RETURN (SAVE) CHARSET X-UNKNOWN ALL
FETCH $ UID
SEARCH RETURN (SAVE) ALL
SEARCH RETURN (SAVE) FOOBAR
FETCH $ UID
STORE 1 +FLAGS.SILENT (\Deleted)
EXPUNGE
FETCH $ UID
SELECT INBOX
FETCH $ UID
SEARCH MODSEQ 1
SEARCH RETURN (MIN COUNT) MODSEQ "/flags/\\draft" all 1
SEARCH MODSEQ "/flags/\\recent" all 1
