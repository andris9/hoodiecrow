# MULTIAPPEND, CATENATE, URL-PARTIAL and REPLACE on the default storage
# Run with: npm run compare -- --plugin UIDPLUS,MULTIAPPEND,CATENATE,REPLACE compare/scenarios/append-extensions.txt
SELECT INBOX
# MULTIAPPEND (RFC 3502), APPENDUID with a UID set (RFC 4315 section 3)
APPEND INBOX (\Seen) {5}\r\nhello (\Flagged) " 7-Feb-1994 22:43:04 -0800" {5}\r\nworld
# a zero-length literal cancels the whole APPEND
APPEND INBOX {5}\r\nhello {0}\r\n
UID FETCH 1:* (FLAGS RFC822.SIZE)
# CATENATE (RFC 4469) with absolute-path URLs
APPEND INBOX CATENATE (URL "/INBOX/;UID=1/;SECTION=HEADER" TEXT {6}\r\nbody\r\n)
APPEND INBOX CATENATE (URL "/INBOX/;UID=3/;SECTION=3.HEADER.FIELDS%20(Subject)" URL "/INBOX/;UID=1/;SECTION=1/;PARTIAL=2.3")
UID FETCH 6:* BODY.PEEK[]
APPEND INBOX CATENATE (URL "/INBOX/;UID=99")
APPEND INBOX CATENATE (URL "/Nope/;UID=1")
APPEND INBOX CATENATE (URL "/INBOX;UID=1")
APPEND INBOX CATENATE ()
# CATENATE inside MULTIAPPEND
APPEND INBOX {5}\r\nhello CATENATE (TEXT {5}\r\nworld)
# REPLACE (RFC 8508)
REPLACE 1 INBOX (\Seen) {5}\r\nhello
UID REPLACE 2 INBOX CATENATE (URL "/INBOX/;UID=3/;SECTION=1" TEXT {3}\r\nabc)
UID REPLACE 99 INBOX {5}\r\nhello
REPLACE 99 INBOX {5}\r\nhello
REPLACE 1 Nope {5}\r\nhello
REPLACE 1 INBOX {5}\r\nhello {5}\r\nworld
UID FETCH 1:* (FLAGS RFC822.SIZE)
