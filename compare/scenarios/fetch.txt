# FETCH, SEARCH and STORE on the default storage
SELECT INBOX
FETCH 1:* (UID FLAGS INTERNALDATE RFC822.SIZE)
FETCH * FLAGS
FETCH 1:* ENVELOPE
FETCH 1:* BODYSTRUCTURE
FETCH 3 (BODY.PEEK[TEXT] BODY.PEEK[3.HEADER] BODY.PEEK[3.MIME] BODY.PEEK[2])
FETCH 1 (BODY.PEEK[HEADER.FIELDS (Subject From)] BODY.PEEK[]<0.10>)
FETCH 1:* (BODY[1])
SEARCH UNSEEN
SEARCH SUBJECT message
SEARCH OR FLAGGED SEEN
SEARCH SENTON 6-Oct-2026
SEARCH LARGER 100000
STORE 2 +FLAGS (\Seen)
STORE 2 -FLAGS.SILENT (\Flagged)
UID FETCH 1:* FLAGS
APPEND INBOX (\Seen) "06-Oct-2026 13:00:05 +0300" {file:../messages/simple.eml}
NOOP
